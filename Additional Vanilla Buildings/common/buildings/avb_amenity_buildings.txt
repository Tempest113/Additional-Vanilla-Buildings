building_avb_residence_3 = {
	base_buildtime = @avb_b3_time
	can_build = no

	category = amenity

	building_sets = {
		government
		urban
		entertainment
		resort
	}

	potential = {
		OR = {
			NOT = { is_planet_class = pc_habitat }
			AND = {
				owner? = {
					is_void_dweller_empire = yes
				}
			}
		}
		NOR = {
			has_modifier = slave_colony
			has_modifier = penal_colony
		}
		owner? = {
			OR = {
				NOT = { has_origin = origin_forever_cruise }
				has_country_flag = forever_cruise_over
			}
		}
	}

	allow = {
		has_major_upgraded_capital = yes
	}

	destroy_trigger = {
		exists = owner
		OR = {
			owner = { is_regular_empire = no }
			owner = { has_valid_civic = civic_shared_burden }
			owner = { has_origin = origin_forever_cruise }
			has_modifier = slave_colony
		}
	}

	convert_to = {
		building_avb_organic_sanctuary_3
		building_avb_drone_storage_3
		building_avb_warren_3
		building_avb_communal_housing_3
	}

	planet_modifier = {
		planet_housing_add = 6000
		planet_amenities_add = 6000
	}

	resources = {
		category = planet_buildings
		inline_script = {
			script = buildings/nomadic_cost_switcher
			REGULAR_RESOURCE = minerals
			NOMADIC_RESOURCE = alloys
			COST = @avb_b3_minerals
			NOMADIC_COST_MULT = @nomadic_minerals_to_strategic_ratio
		}
		cost = {
			rare_crystals = @avb_b3_rare_cost
		}
		upkeep = {
			energy = @avb_b3_upkeep
			rare_crystals = @avb_b3_rare_upkeep
		}
	}

	prerequisites = {
		"tech_avb_residence_3"
	}

	upgrades = {
		"building_avb_residence_4"
	}

	show_tech_unlock_if = {
		owner = {
			is_regular_empire = yes
			NOT = { has_valid_civic = civic_shared_burden }
		}
	}

	ai_weight_coefficient = 1.25
}
