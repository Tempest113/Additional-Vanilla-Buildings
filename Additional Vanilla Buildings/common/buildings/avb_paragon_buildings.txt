building_avb_paragon_memory_vaults_2 = {
	base_buildtime = @avb_b3_time
	can_build = no
	category = unity
	can_demolish = yes
	can_be_ruined = yes
	can_be_disabled = yes
	is_essential = yes

	building_sets = {
		government
		urban
		unity
	}

	destroy_trigger = {
		owner? = {
			is_memory_vault_empire = no
		}
	}

	desc = {
		trigger = {
			owner? = {
				OR = {
					has_civic = civic_memory_vault

				}
			}
		}
		text = building_paragon_memory_vaults_desc
	}

	desc = {
		trigger = {
			owner? = { has_civic = civic_memory_vault_machine }
		}
		text = building_paragon_memory_vaults_machine_desc
	}

	desc = {
		trigger = {
			owner? = { has_civic = civic_memory_vault_hive }
		}
		text = building_paragon_memory_vaults_hive_desc
	}

	triggered_desc = {
		text = memory_vault_tooltip
	}

	allow = {
		has_upgraded_capital_4 = yes
		hidden_trigger = {
			owner = {
				OR = {
					is_ai = no
					AND = {
						resource_stockpile_compare = { resource = minerals value >= 1200 }
						resource_stockpile_compare = { resource = rare_crystals value >= 400 }
					}
				}
			}
		}
	}

	potential = {
		custom_tooltip = {
			text = EMPIRE_UNIQUE_BUILDING
			owner = { is_memory_vault_empire = yes }
		}
		exists = planet
	}

	resources = {
		category = planet_buildings
		inline_script = {
			script = buildings/nomadic_cost_switcher

			NOMADIC_RESOURCE = alloys
			COST = @avb_b3_minerals
			NOMADIC_COST_MULT = @nomadic_minerals_to_strategic_ratio
		}
		cost = {
			rare_crystals = @avb_b3_rare_cost
		}
		cost = {
			trigger = {
				owner? = {
					is_wilderness_empire = yes
				}
			}
			biomass = 800
		}
		upkeep = {
			energy = @avb_b3_upkeep
			rare_crystals = @avb_b2_rare_upkeep
		}
		upkeep = {
			trigger = {
				owner? = { has_civic = civic_memory_vault_corporate }
				owner? = { is_variable_set = paragon_memory_vault }
			}
			energy = @avb_b3_upkeep
			rare_crystals = @avb_b2_rare_upkeep
			multiplier = owner.paragon_memory_vault
		}
	}

	triggered_country_modifier = {
		potential = {
			owner? = {
				is_memory_vault_empire = yes
			}
		}
		species_leader_exp_gain = 0.015
	}

	triggered_country_modifier = {
		potential = {
			owner? = {
				is_memory_vault_empire = yes
				is_variable_set = paragon_memory_vault
			}
		}
		species_leader_exp_gain = 0.015
		mult = owner.paragon_memory_vault
	}

	triggered_country_modifier = {
		potential = {
			owner? = {
				is_memory_vault_empire = yes
			}
		}
		country_unity_produces_mult = 0.015
	}

	triggered_country_modifier = {
		potential = {
			owner? = {
				is_memory_vault_empire = yes
				is_variable_set = paragon_memory_vault
			}
		}
		country_unity_produces_mult = 0.015
		mult = owner.paragon_memory_vault
	}

	inline_script = {
		script = buildings/on_all_wilderness_buildings_districts
	}

	prerequisites = {
		"tech_avb_paragon_memory_vaults_2"
	}
}
