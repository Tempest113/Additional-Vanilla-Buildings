building_avb_monument_4 = {
	base_buildtime = @avb_b4_time
	can_build = no

	category = unity

	building_sets = {
		government
		unity
	}

	potential = {
		NOR = {
			has_modifier = slave_colony
			has_modifier = penal_colony
		}
		owner? = {
			NOR = {
				is_gestalt = yes
				is_megacorp = yes
				is_memorialist_empire = yes
				is_entropy_drinkers_empire = yes
			}
		}
	}

	allow = {
		has_fully_upgraded_capital = yes
	}

	destroy_trigger = {
		exists = owner
		OR = {
			has_modifier = slave_colony
			has_modifier = penal_colony
			owner = {
				OR = {
					is_gestalt = yes
					is_megacorp = yes
					is_memorialist_empire = yes
					is_entropy_drinkers_empire = yes
				}
			}
		}
	}

	convert_to = {
		building_avb_monument_corp_4
		building_avb_monument_hive_4
		building_avb_monument_machine_4
		building_avb_monument_memorial_4
		building_avatar_chamber_4
	}

	resources = {
		category = planet_buildings
		produces = {
			trigger = { exists = owner }
			unity = 4
			multiplier = owner.trigger:num_ascension_perks
		}
		inline_script = {
			script = buildings/nomadic_cost_switcher
			REGULAR_RESOURCE = minerals
			NOMADIC_RESOURCE = rare_crystals
			COST = @avb_b4_minerals
			NOMADIC_COST_MULT = @nomadic_minerals_to_strategic_ratio
		}
	}

	planet_modifier = {
		planet_jobs_unity_produces_mult = 0.20
		planet_civilians_unity_produces_add = 0.1
		planet_civilians_consumer_goods_upkeep_add = 0.1
	}

	triggered_desc = {
		text = building_avb_monument_effect_4
	}

	triggered_desc = {
		text = job_civilian_ethics_effect
	}

	prerequisites = {
		"tech_avb_monument_4"
	}

	upgrades = {
		building_avb_monument_5
	}

	show_tech_unlock_if = {
		is_gestalt = no
		is_megacorp = no
		is_memorialist_empire = no
		is_entropy_drinkers_empire = no
	}
}
