building_avb_automation_3 = {
	base_buildtime = @avb_b3_time
	can_build = no
	category = pop_assembly

	building_sets = {
		government
		urban
		farming
		mining
		generator
	}

	potential = {
		always = no
	}

	allow = {
		always = no
	}

	destroy_trigger = { }

	planet_modifier = {
		district_automated_workforce = 0.75
	}

	resources = {
		category = planet_buildings
		inline_script = {
			script = buildings/nomadic_cost_switcher
			REGULAR_RESOURCE = minerals
			NOMADIC_RESOURCE = rare_crystals
			COST = @avb_b4_minerals
			NOMADIC_COST_MULT = @nomadic_minerals_to_strategic_ratio
		}
		cost = {
			rare_crystals = @avb_b3_rare_cost
			exotic_gases = @avb_b3_rare_cost
			volatile_motes = @avb_b3_rare_cost
		}
		upkeep = {
			energy = @b4_upkeep
			rare_crystals = @avb_b3_rare_upkeep
			exotic_gases = @avb_b3_rare_upkeep
			volatile_motes = @avb_b3_rare_upkeep
		}
	}

	prerequisites = {
		tech_avb_construction_templates_3
	}

	upgrades = {
		building_avb_automation_4
	}

	show_tech_unlock_if = {
		is_tankbound_empire = no
		is_wilderness_empire = no
	}

	triggered_desc = {
		text = avb_automation_3_tooltip
	}
}
