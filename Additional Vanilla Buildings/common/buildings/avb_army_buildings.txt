building_avb_military_academy_2 = {
	base_buildtime = @avb_b3_time
	can_build = no

	can_be_ruined = no
	category = army

	building_sets = {
		fortress
	}

	potential = {
		owner? = {
			is_wilderness_empire = no
			NOT = { has_valid_civic = civic_reanimated_armies }
		}
		NOR = {
			has_modifier = resort_colony
			has_modifier = penal_colony
		}
	}

	allow = {
		OR = {
			has_upgraded_capital_4 = yes
			AND = {
				has_major_upgraded_capital = yes
				owner? = {
					has_valid_civic = civic_machine_tactical_algorithms
				}
			}
		}
	}

	triggered_country_modifier = {
		potential = {
			owner? = { has_valid_civic = civic_machine_tactical_algorithms }
		}
		ship_evasion_mult = 0.03
		ship_tracking_mult = 0.015
		ship_shield_penetration_add = 0.075
		ship_armor_penetration_add = 0.075
		mult = owner.value:tactical_algorithms_scaling_mult
	}

	triggered_country_modifier = {
		potential = {
			owner? = { has_valid_civic = civic_machine_tactical_algorithms }
		}
		ship_shield_hardening_add = 0.15
		mult = owner.value:tactical_algorithms_enclave_scaling_mult
	}

	triggered_country_modifier = {
		potential = {
			owner? = {
				is_gestalt = no
			}
		}
		pop_soldier_bonus_workforce_mult = 0.225
	}

	triggered_country_modifier = {
		potential = {
			owner? = {
				is_gestalt = yes
			}
		}
		pop_warrior_drone_bonus_workforce_mult = 0.225
	}

	triggered_country_modifier = {
		potential = {
			owner? = {
				has_origin = origin_toxic_knights
			}
		}
		pop_knight_bonus_workforce_mult = 0.075
		pop_knight_commander_bonus_workforce_mult = 0.0375
	}

	resources = {
		category = planet_buildings
		inline_script = {
			script = buildings/nomadic_cost_switcher
			REGULAR_RESOURCE = minerals
			NOMADIC_RESOURCE = volatile_motes
			COST = @avb_b3_minerals
			NOMADIC_COST_MULT = @nomadic_minerals_to_strategic_ratio
		}
		upkeep = {
			energy = @avb_b3_upkeep
			volatile_motes = @avb_b3_rare_upkeep
		}
		# Machines with civic
		cost = {
			trigger = {
				owner? = {
					NOT = { has_valid_civic = civic_machine_tactical_algorithms }
				}
			}
			volatile_motes = @avb_b3_rare_cost
		}
		upkeep = {
			trigger = {
				owner? = {
					NOT = { has_valid_civic = civic_machine_tactical_algorithms }
				}
			}
			volatile_motes = @avb_b3_rare_upkeep
		}
		# Machines with civic
		cost = {
			trigger = {
				owner? = {
					has_valid_civic = civic_machine_tactical_algorithms
				}
			}
			volatile_motes = @b2_rare_cost
		}
		upkeep = {
			trigger = {
				owner? = {
					has_valid_civic = civic_machine_tactical_algorithms
				}
			}
			volatile_motes = @b2_rare_upkeep
		}
	}

	prerequisites = {
		"tech_avb_military_academy_2"
	}

	inline_script = {
		script = jobs/soldiers_add
		AMOUNT = @avb_b2_jobs
	}

	show_tech_unlock_if = {
		always = yes
	}
}
