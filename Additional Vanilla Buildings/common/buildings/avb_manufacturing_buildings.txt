building_avb_foundry_4 = {
	base_buildtime = @avb_b3_time
	category = manufacturing
	can_build = no

	building_sets = {
		foundry
		industrial
	}

	potential = {
		owner? = { is_wilderness_empire = no } # see 'building_natural_furnace_3'
		is_special_colony_type = no
	}

	allow = {
		has_fully_upgraded_capital = yes
	}

	destroy_trigger = {
		exists = owner
		OR = {
			has_modifier = resort_colony
			has_modifier = slave_colony
			owner = { is_wilderness_empire = yes }
		}
	}

	resources = {
		category = planet_buildings
		inline_script = {
			script = buildings/nomadic_cost_switcher
			REGULAR_RESOURCE = minerals
			NOMADIC_RESOURCE = volatile_motes
			COST = @avb_b3_minerals
			NOMADIC_COST_MULT = @nomadic_minerals_to_strategic_ratio
		}
		cost = {
			volatile_motes = @avb_b5_rare_cost
		}
		upkeep = {
			energy = @avb_b4_upkeep
			volatile_motes = @avb_b5_rare_upkeep
		}
	}

	prerequisites = {
		"tech_avb_alloys_3"
	}

	inline_script = {
		script = jobs/foundry_add
		AMOUNT = @avb_b4_jobs
	}

	upgrades = {
		building_avb_foundry_5
	}
}
